package db.migration.h2;

import org.flywaydb.core.api.migration.BaseJavaMigration;
import org.flywaydb.core.api.migration.Context;
import java.sql.*;
import java.util.*;

/** Extend the existing menu source; never create pages, categories or posts from IA labels. */
public class V4__navigation extends BaseJavaMigration {
    @Override public void migrate(Context context) throws Exception {
        Connection db = context.getConnection();
        var constraints = new ArrayList<String>();
        try (var statement = db.createStatement(); var rows = statement.executeQuery("SELECT t.CONSTRAINT_NAME FROM INFORMATION_SCHEMA.TABLE_CONSTRAINTS t JOIN INFORMATION_SCHEMA.CHECK_CONSTRAINTS c ON t.CONSTRAINT_SCHEMA=c.CONSTRAINT_SCHEMA AND t.CONSTRAINT_NAME=c.CONSTRAINT_NAME WHERE t.TABLE_NAME='SITE_MENUS' AND c.CHECK_CLAUSE LIKE '%KIND%'")) {
            while (rows.next()) constraints.add(rows.getString(1));
        }
        try (var statement = db.createStatement()) {
            for (String name : constraints) statement.execute("ALTER TABLE site_menus DROP CONSTRAINT \"" + name.replace("\"", "\"\"") + "\"");
            statement.execute("ALTER TABLE site_menus ADD CONSTRAINT site_menu_kind CHECK (kind IN ('PAGE','CATEGORY','LINK','GROUP'))");
            statement.execute("ALTER TABLE site_menus ADD COLUMN parent_id BIGINT REFERENCES site_menus(id)");
            statement.execute("CREATE INDEX site_menu_parent ON site_menus(parent_id,sort_order)");
        }
        for (String label : List.of("인사교 알아보기", "선배들의 SSUL", "인사교 Real Life", "지원 전 Check!!", "FAQ")) {
            try (var statement = db.prepareStatement("INSERT INTO site_menus(label,kind,sort_order,visible) SELECT ?,'GROUP',COALESCE(MAX(sort_order),0)+1,FALSE FROM site_menus HAVING NOT EXISTS (SELECT 1 FROM site_menus WHERE label=?)")) {
                statement.setString(1,label); statement.setString(2,label); statement.executeUpdate();
            }
        }
        // Only an existing, unambiguous introduction menu is grouped. Other targets await explicit mapping.
        try (var statement = db.createStatement()) {
            statement.executeUpdate("UPDATE site_menus SET parent_id=(SELECT MIN(id) FROM site_menus WHERE kind='GROUP' AND label='인사교 알아보기') WHERE parent_id IS NULL AND kind='PAGE' AND target_id IN (SELECT id FROM site_pages WHERE title IN ('인사교 소개','인사교 간단 소개','사관학교 소개'))");
            // A container above an already visible menu must not silently hide that existing destination.
            statement.executeUpdate("UPDATE site_menus SET visible=TRUE WHERE kind='GROUP' AND id IN (SELECT parent_id FROM site_menus WHERE visible=TRUE AND parent_id IS NOT NULL)");
        }
    }
}
